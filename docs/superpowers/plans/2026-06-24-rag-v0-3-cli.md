# RAG 知识库 v0.3 CLI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为 `rag-knowledge-base` 增加可日常使用的命令行入口，让用户不用写 Python 代码也能完成 ingest / query / ask / list / stats / delete / rebuild。

**Architecture:** CLI 是 `KnowledgeBase` 的薄壳，不放业务逻辑。先在核心层补齐文档管理方法，再用 `argparse` 实现 `cli.py`，所有命令共享 `create_default_kb()` / `KnowledgeBase.from_config()`。默认输出人类可读文本，`--json` 输出机器可读 JSON。

**Tech Stack:** Python 3.12、标准库 `argparse` / `json` / `pathlib` / `shutil`、pytest、现有 `rag` 包。

## v0.3 范围

实现以下命令：

```bash
python cli.py ingest <path> [--recursive] [--db-path data/knowledge.db] [--json]
python cli.py query "问题" [--top-k 5] [--db-path data/knowledge.db] [--json]
python cli.py ask "问题" [--top-k 5] [--db-path data/knowledge.db] [--json]
python cli.py list [--db-path data/knowledge.db] [--json]
python cli.py stats [--db-path data/knowledge.db] [--json]
python cli.py delete <doc_id_or_source> [--db-path data/knowledge.db] [--json]
python cli.py rebuild [--db-path data/knowledge.db] [--yes] [--json]
```

不做：

- 不接真实 LLM。
- 不接真实 semantic embedding。
- 不做 REST API。
- 不做 Web UI。
- 不做 PDF / Word / URL。

```python
    def list_docs(self) -> list[Document]:
        """列出知识库内文档。"""
        return self._store.list_docs()

    def stats(self) -> dict:
        """返回文档数、chunk 数和向量维度。"""
        return {
            "documents": len(self._store.list_docs()),
            "chunks": self._store.count(),
            "dim": self.embedder.dim,
        }

    def delete(self, target: str) -> dict:
        """按 doc_id 或 source 删除文档。优先匹配 doc_id，再匹配 source。"""
        for doc in self._store.list_docs():
            if doc.doc_id == target:
                return {"deleted": self._store.delete_doc(doc.doc_id), "target": target}
        deleted = self._store.delete_by_source(target)
        return {"deleted": deleted, "target": target}

    def rebuild(self) -> dict:
        """清空当前知识库。v0.3 只负责清空，不自动重新摄入原始目录。"""
        self._store.clear()
        return {"documents": 0, "chunks": 0}
```

```python
from __future__ import annotations

import argparse
import json
import sys
from dataclasses import asdict, is_dataclass
from pathlib import Path
from typing import Any

from rag import KnowledgeBase
from rag.errors import KnowledgeBaseError


def _json_default(value: Any):
    if is_dataclass(value):
        return asdict(value)
    raise TypeError(f"{type(value).__name__} is not JSON serializable")


def print_json(payload: Any) -> None:
    print(json.dumps(payload, ensure_ascii=False, indent=2, default=_json_default))


def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(description="RAG 知识库命令行工具")
    parser.add_argument("--db-path", default="data/knowledge.db", help="SQLite 知识库路径")
    parser.add_argument("--json", action="store_true", help="输出 JSON")
    sub = parser.add_subparsers(dest="command", required=True)

    ingest = sub.add_parser("ingest", help="摄入文件或目录")
    ingest.add_argument("path")
    ingest.add_argument("--recursive", action="store_true")

    query = sub.add_parser("query", help="只检索片段，不调用生成器")
    query.add_argument("question")
    query.add_argument("--top-k", type=int, default=5)

    ask = sub.add_parser("ask", help="检索并生成答案")
    ask.add_argument("question")
    ask.add_argument("--top-k", type=int, default=5)

    sub.add_parser("list", help="列出文档")
    sub.add_parser("stats", help="显示统计信息")

    delete = sub.add_parser("delete", help="按 doc_id 或 source 删除")
    delete.add_argument("target")

    rebuild = sub.add_parser("rebuild", help="清空知识库")
    rebuild.add_argument("--yes", action="store_true", help="确认清空")

    return parser


def main(argv: list[str] | None = None) -> int:
    parser = build_parser()
    args = parser.parse_args(argv)
    kb = KnowledgeBase.from_config(db_path=args.db_path)

    try:
        if args.command == "query":
            result = kb.query(args.question, top_k=args.top_k)
            if args.json:
                print_json(result)
            else:
                for i, item in enumerate(result, 1):
                    print(f"[{i}] score={item.score:.4f} source={item.chunk.source}")
                    print(item.chunk.text)
            return 0

        parser.error(f"未实现的命令：{args.command}")
    except KnowledgeBaseError as exc:
        print(str(exc), file=sys.stderr)
        return 1


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
SUPPORTED_EXTENSIONS = {".md", ".txt"}


def _iter_sources(path: Path, recursive: bool) -> tuple[list[Path], list[Path]]:
    if path.is_file():
        return ([path], []) if path.suffix.lower() in SUPPORTED_EXTENSIONS else ([], [path])
    if path.is_dir():
        if not recursive:
            raise ValueError("目录摄入需要 --recursive")
        files = [p for p in path.rglob("*") if p.is_file()]
        good = [p for p in files if p.suffix.lower() in SUPPORTED_EXTENSIONS]
        skipped = [p for p in files if p.suffix.lower() not in SUPPORTED_EXTENSIONS]
        return good, skipped
    raise FileNotFoundError(str(path))
```
